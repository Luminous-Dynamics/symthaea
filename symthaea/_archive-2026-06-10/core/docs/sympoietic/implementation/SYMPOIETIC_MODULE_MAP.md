}

impl PartnershipContext {
    /// Record an interaction and update relationship
    pub fn record_interaction(
        &mut self,
        time: f64,
        my_state: &[f32],
        partner_state: &[f32],
        speaker: Option<&str>,
    ) {
        // Convert to HV16 and record
        self.relational.interact(time, my_hv, partner_hv, speaker.map(String::from));
    }

    /// Get current relationship assessment
    pub fn assess(&self) -> RelationalAssessment {
        self.relational.assess()
    }

    /// Check if partner needs support
    pub fn needs_support(&self) -> bool {
        self.vulnerability.is_stressed ||
        self.vulnerability.needs_emotional_support
    }
}

/// Partnership-weighted attention bid
pub struct PartnershipBid {
    pub bid: AttentionBid,
    pub partner_vulnerability: f32,
    pub care_type: SupportType,
    pub reciprocity_boost: f32,
}

impl PartnershipBid {
    pub fn partnership_score(&self) -> f32 {
        let base = self.bid.score();
        let vulnerability_weight = 1.0 + (self.partner_vulnerability * 0.5);
        let reciprocity_weight = 1.0 + (self.reciprocity_boost * 0.3);
        base * vulnerability_weight * reciprocity_weight
    }
}
