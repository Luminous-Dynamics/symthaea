```rust
/// Hierarchical Φ for networks
pub struct NetworkPhiCalculator {
    /// Number of levels in hierarchy
    levels: usize,

    /// Community detection algorithm
    community_algorithm: CommunityDetection,
}

impl NetworkPhiCalculator {
    /// Compute Φ at network level
    pub fn compute_network_phi(&self, network: &ConsciousnessNetwork) -> NetworkPhi {
        // 1. Detect natural communities (hierarchical clustering)
        let communities = self.detect_communities(network);

        // 2. Compute Φ at each level
        let mut level_phis = Vec::new();

        // Level 0: Individual Φ values
        let individual_phis: Vec<f64> = network.nodes.iter()
            .map(|n| n.phi_individual)
            .collect();
        level_phis.push(LevelPhi::individual(individual_phis));

        // Level 1: Dyadic Φ values
        let dyad_phis: Vec<f64> = network.edges.iter()
            .map(|e| e.phi_dyad)
            .collect();
        level_phis.push(LevelPhi::dyadic(dyad_phis));

        // Level 2+: Community Φ values
        for level in 0..communities.levels() {
            let community_phis = communities.at_level(level).iter()
                .map(|c| self.compute_community_phi(c, network))
                .collect();
            level_phis.push(LevelPhi::community(level, community_phis));
        }

        // 3. Integrate across levels
        NetworkPhi {
            level_phis,
            integrated: self.integrate_levels(&level_phis),
        }
    }

    /// Community Φ uses the "information geometry" approach
    fn compute_community_phi(
        &self,
        community: &Community,
        network: &ConsciousnessNetwork
    ) -> f64 {
        // Information that exists in the community but not in subparts
        let whole = self.mutual_information_whole(community, network);
        let parts = self.mutual_information_parts(community, network);

        // Φ = information generated by integration
        whole - parts
    }
}
```

```rust
    /// Is the network at criticality?
    pub fn is_critical(&self) -> bool {
        // At criticality:
        // 1. Lyapunov ≈ 0 (edge of chaos)
        // 2. Power-law exponent 2 < α < 3 (scale-free)
        // 3. Correlation length diverges
        // 4. High susceptibility

        let lyap_ok = self.lyapunov.abs() < 0.1;
        let power_ok = self.power_law_alpha > 2.0 && self.power_law_alpha < 3.0;
        let corr_ok = self.correlation_length > 10.0;  // Relative to network size
        let susc_ok = self.susceptibility > 5.0;

        lyap_ok && power_ok && corr_ok && susc_ok
    }

    /// Compute Lyapunov exponent from network dynamics
    fn compute_lyapunov(&self, network: &ConsciousnessNetwork) -> f64 {
        // Perturb the network slightly
        let mut perturbed = network.clone();
        let epsilon = 1e-6;

        // Add small random perturbations
        for node in &mut perturbed.nodes {
            node.phi_individual += epsilon * random_normal();
        }

        // Evolve both networks
        let original_evolved = evolve(network, steps: 100);
        let perturbed_evolved = evolve(&perturbed, steps: 100);

        // Measure divergence
        let divergence = distance(&original_evolved, &perturbed_evolved);

        // Lyapunov = log(divergence / epsilon) / steps
        (divergence / epsilon).ln() / 100.0
    }
```

```rust
                    edges.push(DyadicEdge::new(i, j));
                }
            }
        }

        ConsciousnessNetwork::new(nodes.to_vec(), edges)
```

### Beyond Pairwise Consciousness

Dyads are pairwise. But what about triads, tetrads, and N-way relationships?

```rust
/// Hypergraph model for multi-party consciousness
pub struct ConsciousnessHypergraph {
    /// Nodes (individual consciousnesses)
    nodes: Vec<ConsciousnessNode>,

    /// Hyperedges (N-way relationships)
    hyperedges: Vec<Hyperedge>,
}

pub struct Hyperedge {
    /// Participants in this N-way relationship
    participants: Vec<NodeId>,

    /// Φ of this N-way relationship
    phi: f64,

    /// Quality metrics
    coherence: f64,
    shared_meaning: f64,
}

impl ConsciousnessHypergraph {
    /// Compute N-way Φ
    pub fn compute_hyperedge_phi(&self, edge: &Hyperedge) -> f64 {
        let n = edge.participants.len();

        // Generalization of dyadic Φ to N parties
        // More complex than pairwise because partitions grow combinatorially

        match n {
            2 => self.dyadic_phi(edge),
            3 => self.triadic_phi(edge),
            4 => self.tetradic_phi(edge),
            _ => self.general_n_phi(edge),
        }
    }

    /// Triadic Φ (three-way consciousness)
    fn triadic_phi(&self, edge: &Hyperedge) -> f64 {
        let [a, b, c] = [edge.participants[0], edge.participants[1], edge.participants[2]];

        // Three possible 2-1 partitions
        let phi_ab_c = self.partition_phi(&[a, b], &[c]);
        let phi_ac_b = self.partition_phi(&[a, c], &[b]);
        let phi_bc_a = self.partition_phi(&[b, c], &[a]);

        // Triadic Φ = minimum information lost in any partition
        phi_ab_c.min(phi_ac_b).min(phi_bc_a)
    }
}
```
