        assert_eq!(stats.accuracy, 0.5);
        assert_eq!(stats.source_count, 0);
        assert_eq!(stats.domain_count, 0);
        assert!(!learner.is_ready());
        assert_eq!(
            learner.get_domain_expertise("science"),
            0.5,
            "unadjudicated predictions must not train domain expertise"
        );
        assert!(
            learner
                .get_source_credibility("self-referential.example")
                .is_none(),
            "unadjudicated predictions must not train source credibility"
        );
    }

        let mut learner = EpistemicLearner::new();
        for index in 0..9 {
            learner
                .record_outcome(make_outcome(

                .unwrap();

    }

    #[test]
    fn pending_outcomes_cannot_evict_adjudicated_calibration_history() {
        let mut learner = EpistemicLearner::new();
        for index in 0..10 {
            learner
                .record_outcome(make_outcome(
                    "science",
                    "adjudicated.example",
                    index % 2 == 0,
                ))
                .unwrap();
        }
        for index in 0..1000 {
            learner
                .record_outcome(VerificationOutcome::unadjudicated(
                    format!("Pending claim {index}"),
                    "pending.example",
                    ResearchSource::Web,
                ))
                .unwrap();
        }

        let stats = learner.stats();
        assert_eq!(stats.total_outcomes, 1000);
        assert_eq!(stats.adjudicated_outcomes, 10);
        assert_eq!(stats.unadjudicated_outcomes, 990);
        assert!(learner.is_ready());
        assert_eq!(stats.accuracy, 0.5);
        assert!(
            learner
                .get_source_credibility("adjudicated.example")
                .is_some(),
            "pending records must not evict the measured source calibration"
        );
    }

    #[test]
    fn accuracy_uses_only_independently_adjudicated_outcomes() {
        let mut learner = EpistemicLearner::new();
        learner
            .record_outcome(make_outcome("science", "source.example", true))
            .unwrap();
        learner
            .record_outcome(VerificationOutcome::unadjudicated(

                "source.example",

            .unwrap();

        assert_eq!(stats.adjudicated_outcomes, 2);
        assert_eq!(stats.unadjudicated_outcomes, 1);
        assert_eq!(stats.accuracy, 0.5);
    }

    #[test]
    fn test_source_credibility_learning() {
        let mut learner = EpistemicLearner::new();

        // Source that is mostly correct
        for _ in 0..20 {
            learner
                .record_outcome(make_outcome("programming", "docs.rust-lang.org", true))
                .unwrap();
        }

        // Source that is often wrong
        for _ in 0..20 {
            learner
                .record_outcome(make_outcome("programming", "random-blog.com", false))
                .unwrap();
        }

        let good_source = learner
            .get_source_credibility("docs.rust-lang.org")
            .unwrap();
        let bad_source = learner.get_source_credibility("random-blog.com").unwrap();

        assert!(good_source.credibility > bad_source.credibility);
        assert!(good_source.accuracy() > 0.9);
        assert!(bad_source.accuracy() < 0.1);
    }

    fn test_recommend_sources() {
        let mut learner = EpistemicLearner::new();

        // Build history with different sources
        for _ in 0..10 {
            learner
                .record_outcome(make_outcome("programming", "rust-lang.org", true))
                .unwrap();

        for _ in 0..10 {
            learner
                .record_outcome(make_outcome("programming", "stackoverflow.com", true))

                .unwrap();

        // rust-lang.org should be ranked higher
        if recommendations.len() >= 2 {
            assert!(recommendations[0].0 == "rust-lang.org" || recommendations[0].1 > 0.7);
        }
    }
