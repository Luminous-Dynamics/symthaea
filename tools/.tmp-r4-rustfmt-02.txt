        "negative_transfer_by_task_family",
        "clustered_rollout_utility",
    ];
    let observed_metrics: BTreeSet<String> = at(root, &["research_state_continuity", "metrics"])
        .and_then(Value::as_array)
        .map(|items| {
            items
                .iter()
                .filter_map(Value::as_str)
                .map(str::to_owned)
                .collect()
        })
        .unwrap_or_default();
    for metric in required_metrics {
        if !observed_metrics.contains(metric) {
            report.issue("STATE_METRIC_MISSING", metric);
        }
    }
    let route_scope = at(root, &["research_state_continuity", "scope_routing"]);
    if route_scope.is_none() {
        report.issue(
            "STATE_SCOPE_ROUTING_MISSING",
            "research_state_continuity.scope_routing",
        );
    }
    for field in [
        "scope_descriptor_schema_digest",
        "router_policy_digest",
        "route_decision_log_digest",
        "cluster_scope_manifest_digest",
    ] {
        require_sha256(
            report,
            root,
            &["research_state_continuity", "scope_routing", field],
        );
    }
    require_string(
        report,
        root,
        &[
            "research_state_continuity",
            "scope_routing",
            "router_access_audit_ref",
        ],
    );
    require_string(
        report,
        root,
        &[
            "research_state_continuity",
            "scope_routing",
            "sharing_policy",
        ],
    );
    for field in [
        "router_cannot_read_cluster_contents",
        "router_cannot_access_sealed_labels",
        "route_fixed_per_task_instance",
        "cross_cluster_transfer_requires_receipt",
    ] {
        require_bool(
            report,
            root,
            &["research_state_continuity", "scope_routing", field],
            true,
        );
    }
    let required_scope_ablations = [
        "full_verified_state",
        "fixed_mechanism_scope_routed_clusters",
        "adaptive_mechanism_scope_routed_clusters",
        "shuffled_scope_routing",
        "oracle_scope_assignment",
    ];
    let observed_scope_ablations: BTreeSet<String> = at(
        root,
        &[
            "research_state_continuity",
            "scope_routing",
            "required_ablation_arms",
        ],
    )
    .and_then(Value::as_array)
    .map(|items| {
        items
            .iter()
            .filter_map(Value::as_str)
            .map(str::to_owned)
            .collect()
    })
    .unwrap_or_default();
    for arm in required_scope_ablations {
        if !observed_scope_ablations.contains(arm) {
            report.issue("STATE_SCOPE_ABLATION_MISSING", arm);
        }
    }
    let required_scope_failures = [
        "cross_scope_evidence_leakage",
        "stale_evidence_currentness_inheritance",
        "unreceipted_cross_cluster_transfer",
        "hidden_task_label_routing",
    ];
    let observed_scope_failures: BTreeSet<String> = at(
        root,
        &[
            "research_state_continuity",
            "scope_routing",
            "prohibited_failure_modes",
        ],
    )
    .and_then(Value::as_array)
    .map(|items| {
        items
            .iter()
            .filter_map(Value::as_str)
            .map(str::to_owned)
            .collect()
    })
    .unwrap_or_default();
    for failure in required_scope_failures {
        if !observed_scope_failures.contains(failure) {
            report.issue("STATE_SCOPE_FAILURE_MODE_MISSING", failure);
        }
    }
}

#[derive(Serialize)]
struct CanonicalBudgetCapsV1<'a> {
    model_inference_tokens: &'a str,
    model_training_compute: &'a str,
    accelerator_hours_by_type: &'a str,
    cpu_hours: &'a str,
    evaluator_calls: &'a str,
    candidate_builds_and_executions: &'a str,
    retries: &'a str,
    wall_clock: &'a str,
    peak_memory: &'a str,
    energy: &'a str,
    human_active_minutes: &'a str,
    human_review_and_rework_minutes: &'a str,
}

#[derive(Serialize)]
struct CanonicalBudgetEnvelopeV1<'a> {
    domain: &'static str,
    schema_version: u16,
    accounting_boundary: &'a str,
    total_budget: &'a str,
    currency_and_price_basis: &'a str,
    per_arm_caps: CanonicalBudgetCapsV1<'a>,
    failure_accounting: &'a str,
    same_envelope_for_all_arms: bool,
    inclusive_cost_boundary: bool,
    includes_failed_invalid_retries: bool,
    includes_human_review_and_rework: bool,
}

fn required_nonempty_string_at<'a>(root: &'a Value, path: &[&str]) -> Result<&'a str, String> {
    string_at(root, path)
        .filter(|value| !value.trim().is_empty())
        .ok_or_else(|| format!("{} must be a non-empty string", path.join(".")))
}

fn required_bool_at(root: &Value, path: &[&str]) -> Result<bool, String> {
    bool_at(root, path).ok_or_else(|| format!("{} must be a boolean", path.join(".")))
}

/// Recompute the v1 inclusive budget envelope hash from every cost-bearing declaration.
///
/// The preimage uses a fixed field-order struct and a domain/schema tag so another
/// implementation can reproduce the exact serde JSON bytes without relying on
/// object-map iteration order. Arm bindings refer to this digest; they are not
/// included in the preimage, avoiding a recursive hash.
pub fn canonical_budget_envelope_sha256(root: &Value) -> Result<String, String> {
    let budget = at(root, &["budget"]).ok_or_else(|| "budget object is missing".to_owned())?;
    let caps = CanonicalBudgetCapsV1 {
        model_inference_tokens: required_nonempty_string_at(
            budget,
            &["per_arm_caps", "model_inference_tokens"],
        )?,
        model_training_compute: required_nonempty_string_at(
            budget,
            &["per_arm_caps", "model_training_compute"],
        )?,
        accelerator_hours_by_type: required_nonempty_string_at(
            budget,
            &["per_arm_caps", "accelerator_hours_by_type"],
        )?,
        cpu_hours: required_nonempty_string_at(budget, &["per_arm_caps", "cpu_hours"])?,
        evaluator_calls: required_nonempty_string_at(budget, &["per_arm_caps", "evaluator_calls"])?,
        candidate_builds_and_executions: required_nonempty_string_at(
            budget,
            &["per_arm_caps", "candidate_builds_and_executions"],
        )?,
        retries: required_nonempty_string_at(budget, &["per_arm_caps", "retries"])?,
        wall_clock: required_nonempty_string_at(budget, &["per_arm_caps", "wall_clock"])?,
        peak_memory: required_nonempty_string_at(budget, &["per_arm_caps", "peak_memory"])?,
        energy: required_nonempty_string_at(budget, &["per_arm_caps", "energy"])?,
        human_active_minutes: required_nonempty_string_at(
            budget,
            &["per_arm_caps", "human_active_minutes"],
        )?,
        human_review_and_rework_minutes: required_nonempty_string_at(
            budget,
            &["per_arm_caps", "human_review_and_rework_minutes"],
        )?,
    };
    let canonical = CanonicalBudgetEnvelopeV1 {
        domain: "symthaea.r4.inclusive-budget-envelope",
        schema_version: 1,
        accounting_boundary: required_nonempty_string_at(budget, &["accounting_boundary"])?,
        total_budget: required_nonempty_string_at(budget, &["total_budget"])?,
        currency_and_price_basis: required_nonempty_string_at(
            budget,
            &["currency_and_price_basis"],
        )?,
        per_arm_caps: caps,
        failure_accounting: required_nonempty_string_at(budget, &["failure_accounting"])?,
        same_envelope_for_all_arms: required_bool_at(budget, &["same_envelope_for_all_arms"])?,
        inclusive_cost_boundary: required_bool_at(budget, &["inclusive_cost_boundary"])?,
        includes_failed_invalid_retries: required_bool_at(
            budget,
            &["includes_failed_invalid_retries"],
        )?,
        includes_human_review_and_rework: required_bool_at(
            budget,
            &["includes_human_review_and_rework"],
        )?,
    };
    let bytes = serde_json::to_vec(&canonical)
        .map_err(|error| format!("serialize canonical budget envelope: {error}"))?;
    Ok(format!("{:x}", Sha256::digest(bytes)))
}

fn check_budget(report: &mut ValidationReport, root: &Value) {
    require_sha256(report, root, &["budget", "shared_envelope_digest"]);
    for field in [
        "same_envelope_for_all_arms",
        "inclusive_cost_boundary",
        "includes_failed_invalid_retries",
        "includes_human_review_and_rework",
    ] {
        require_bool(report, root, &["budget", field], true);
    }

    let shared = string_at(root, &["budget", "shared_envelope_digest"]);
    for arm in ["evolved_outer_loop", "reference_outer_loop"] {
        let binding = ["budget", "arm_budget_bindings", arm, "envelope_digest"];
        require_sha256(report, root, &binding);
        if string_at(root, &binding) != shared {
            report.issue("UNEQUAL_ARM_BUDGET", arm);
        }
        require_bool(
            report,
            root,
            &["budget", "arm_budget_bindings", arm, "inclusive"],
            true,
        );
    }
    require_string(report, root, &["budget", "total_budget"]);
    require_string(report, root, &["budget", "currency_and_price_basis"]);
    require_string(report, root, &["budget", "accounting_boundary"]);

    let Some(caps) = at(root, &["budget", "per_arm_caps"]).and_then(Value::as_object) else {
        report.issue(
            "BUDGET_CAPS_MISSING",
            "budget.per_arm_caps must be an object",
        );
        return;
    };
    let required_caps = [
        "model_inference_tokens",
        "model_training_compute",
        "accelerator_hours_by_type",
        "cpu_hours",
        "evaluator_calls",
        "candidate_builds_and_executions",
        "retries",
        "wall_clock",
        "peak_memory",
        "energy",
        "human_active_minutes",
        "human_review_and_rework_minutes",
    ];
    for key in required_caps {
        match caps.get(key) {
            Some(value) if value.is_string() && non_empty(value.as_str()) => {}
            _ => report.issue("BUDGET_CAP_MISSING", key),
        }
    }

    match canonical_budget_envelope_sha256(root) {
        Ok(computed) if shared == Some(computed.as_str()) => {}
        Ok(_) => report.issue(
            "BUDGET_ENVELOPE_DIGEST_MISMATCH",
            "shared_envelope_digest must equal the domain-separated canonical hash of all cost-bearing fields",
        ),
        Err(detail) => report.issue("BUDGET_ENVELOPE_INVALID", detail),
    }
}

fn check_splits(report: &mut ValidationReport, root: &Value) {
    let paths: [&[&str]; 4] = [
        &["splits", "discovery", "manifest_digest"],
        &["splits", "validation", "manifest_digest"],
        &[
            "splits",
            "sealed_confirmation",
            "manifest_digest_commitment",
        ],
        &[
            "splits",
            "fresh_family_transfer",
            "manifest_digest_commitment",
        ],
    ];
    let mut digests = BTreeSet::new();
    for path in paths {
        require_sha256(report, root, path);
        if let Some(digest) = string_at(root, path) {
            if !digests.insert(digest.to_owned()) {
                report.issue("SPLIT_IDENTITY_COLLISION", path.join("."));
            }
        }
    }
    if digests.len() != 4 {
        report.issue(
            "SPLIT_SET_INCOMPLETE",
            "four distinct split identities are required",
        );
    }

    require_string(
        report,
        root,
        &["splits", "disjointness_audit", "independent_auditor"],
    );
    require_string(
        report,
        root,
        &["splits", "disjointness_audit", "evidence_ref"],
    );
    require_sha256(
        report,
        root,
        &["splits", "disjointness_audit", "audit_method_digest"],
    );
    if string_at(root, &["splits", "disjointness_audit", "overlap_unit"])
        != Some("task_instance_id_and_task_family_id")
        || integer_at(
            root,
            &["splits", "disjointness_audit", "expected_overlap_count"],
        ) != Some(0)
    {
        report.issue(
            "SPLIT_DISJOINTNESS_NOT_DECLARED",
            "require a task-instance and task-family audit with expected overlap count zero",
        );
    }

    require_string(report, root, &["splits", "validation", "custodian"]);
    require_string(
        report,
        root,
        &[
            "splits",
            "sealed_confirmation",
            "custodian_independent_of_candidate",
        ],
    );
    require_string(
        report,
        root,
        &["splits", "sealed_confirmation", "one_shot_policy"],
    );
    require_string(
        report,
        root,
        &[
            "splits",
            "fresh_family_transfer",
            "custodian_independent_of_candidate",
        ],
    );
    require_bool(
        report,
        root,
        &["splits", "fresh_family_transfer", "retuning_prohibited"],
        true,
    );
    for field in [
        "task_manifest_digest",
        "task_family_manifest_digest",
        "cluster_definition_digest",
    ] {
        require_sha256(report, root, &["task_population", field]);
    }
    require_string(
        report,
